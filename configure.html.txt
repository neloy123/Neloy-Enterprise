// Load configurations dynamically based on URL parameter
function initConfigurator() {
  const data = productConfigData[productKey] || productConfigData.windows;
  
  document.getElementById("productTitle").innerText = data.title;
  document.getElementById("productSub").innerText = data.sub;
  
  // Render Dynamic Types
  document.getElementById("typeLabel").innerText = data.typeLabel;
  let typeHTML = "";
  data.types.forEach((t, i) => {
    typeHTML += `<button class="option ${i===0?'active':''}" onclick="selectOption(this)"><strong>${t[0]}</strong><span>${t[1]}</span></button>`;
  });
  document.getElementById("typeRow").innerHTML = typeHTML;

  // Render Dynamic Material 1
  document.getElementById("mat1Label").innerText = data.mat1Label;
  let mat1HTML = "";
  data.mat1.forEach((m, i) => {
    mat1HTML += `<button class="option ${i===0?'active':''}" onclick="selectOption(this)"><strong>${m[0]}</strong><span>${m[1]}</span></button>`;
  });
  document.getElementById("mat1Row").innerHTML = mat1HTML;

  // Render Dynamic Material 2
  document.getElementById("mat2Label").innerText = data.mat2Label;
  let mat2HTML = "";
  data.mat2.forEach((m, i) => {
    mat2HTML += `<button class="option ${i===0?'active':''}" onclick="selectOption(this)"><strong>${m[0]}</strong><span>${m[1]}</span></button>`;
  });
  document.getElementById("mat2Row").innerHTML = mat2HTML;

  calculate();
}

function selectOption(element){
  const parent = element.parentElement;
  parent.querySelectorAll(".option").forEach(item => { item.classList.remove("active"); });
  element.classList.add("active");
  calculate();
}

function calculate(){
  const width = parseFloat(document.getElementById("width").value) || 0;
  const height = parseFloat(document.getElementById("height").value) || 0;
  const data = productConfigData[productKey] || productConfigData.windows;
  
  // Active modifier check
  let multiplier = 1;
  const activeOptions = document.querySelectorAll(".option.active");
  activeOptions.forEach(opt => {
    const text = opt.querySelector("strong").innerText.toLowerCase();
    if(text.includes("neo") || text.includes("premium") || text.includes("heavy") || text.includes("cabinet") || text.includes("5 mm") || text.includes("6 mm") || text.includes("with door")) {
      multiplier += 0.25;
    }
  });

  const area = width * height;
  const total = area * data.baseRate * multiplier + 1000; // Base fitting cost inclusion

  document.getElementById("price").innerText = "৳ " + Math.round(total).toLocaleString("en-BD");
  
  // Update WhatsApp link with active choices
  updateWhatsAppLink(width, height, total);
}

function updateWhatsAppLink(w, h, total) {
  const title = document.getElementById("productTitle").innerText;
  let details = [];
  document.querySelectorAll(".option.active").forEach(opt => {
    details.push(opt.querySelector("strong").innerText);
  });
  
  const msg = `Hello Neloy Enterprise, I want to order/inquire about a custom build:\n\n*Product:* ${title}\n*Dimensions:* ${w}ft x ${h}ft\n*Specifications:* ${details.join(", ")}\n*Estimated Price:* ৳ ${Math.round(total).toLocaleString("en-BD")}\n\nPlease confirm availability and final quotation.`;
  
  const encodedMsg = encodeURIComponent(msg);
  document.getElementById("whatsappBtn").href = `https://wa.me/8801722884496?text=${encodedMsg}`;
}

function proceedCheckout() {
  const total = document.getElementById("price").innerText;
  const title = document.getElementById("productTitle").innerText;
  alert(`Thank you! Your configuration for "${title}" with an estimated price of ${total} is ready.\n\nYou will now be redirected to WhatsApp to complete your order with Neloy Enterprise.`);
  window.location.href = document.getElementById("whatsappBtn").href;
}
